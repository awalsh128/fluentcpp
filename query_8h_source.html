704template <typename T>
705Queryable<T> Queryable<T>::difference(std::vector<T> rhs_items) {
706 static_assert(
707 traits::is_less_than_comparable<T>::value,
708 "T must be less-than comparable.");
709 std::sort(rhs_items.begin(), rhs_items.end());
710 // Number of matches consumed so far for each run of equivalent rhs items,
711 // indexed by the start of the run.
712 std::vector<size_t> consumed(rhs_items.size(), 0);
713 std::erase_if(items_, [&](const T &item) {
714 auto [lo, hi] = std::equal_range(rhs_items.begin(), rhs_items.end(), item);
715 size_t &used = consumed[lo - rhs_items.begin()];
716 if (lo + used < hi) {
717 used++;
718 return true;
719 }
720 return false;
721 });
722 return Queryable<T>(std::move(items_));
723}

821template <typename T>
822template <typename U, typename LhsKeySelector, typename RhsKeySelector>
823Queryable<std::tuple<T, U>> Queryable<T>::join(
824 std::vector<U> rhs_items, LhsKeySelector lhs_key_selector,
825 RhsKeySelector rhs_key_selector) {
826 using KU = decltype(rhs_key_selector(*rhs_items.begin()));
827 using KT = decltype(lhs_key_selector(*items_.begin()));
828 static_assert(
829 std::is_same<KU, KT>::value,
830 "Left and right hand key selectors must produce the same type.");
831
832 using K = KT;
833 static_assert(
834 traits::is_less_than_comparable<K>::value,
835 "Key selectors must produce a type that is less-than compareable.");
836
837 std::map<K, std::vector<U>> rhs_mapped;
838 for (U &rhs_item : rhs_items) {
839 rhs_mapped[rhs_key_selector(std::as_const(rhs_item))].push_back(
840 std::move(rhs_item));
841 }
842
843 // Count how many lhs items use each rhs group so the last use can move.
844 std::vector<K> lhs_keys;
845 lhs_keys.reserve(items_.size());
846 std::map<K, size_t> uses_left;
847 for (T &lhs_item : items_) {
848 lhs_keys.push_back(lhs_key_selector(std::as_const(lhs_item)));
849 if (rhs_mapped.count(lhs_keys.back())) {
850 uses_left[lhs_keys.back()]++;
851 }
852 }
853
854 std::vector<std::tuple<T, U>> joined;
855 for (size_t i = 0; i < items_.size(); i++) {
856 auto rhs_it = rhs_mapped.find(lhs_keys[i]);
857 if (rhs_it == rhs_mapped.end()) {
858 continue;
859 }
860 bool last_lhs_use = --uses_left[lhs_keys[i]] == 0;
861 std::vector<U> &rhs_group = rhs_it->second;
862 for (size_t j = 0; j < rhs_group.size(); j++) {
863 bool last_rhs_use = j + 1 == rhs_group.size();
864 joined.emplace_back(
865 last_rhs_use ? std::move(items_[i])
866 : detail::copy_or_throw(items_[i], "join"),
867 last_lhs_use ? std::move(rhs_group[j])
868 : detail::copy_or_throw(rhs_group[j], "join"));
869 }
870 }
871
872 return Queryable<std::tuple<T, U>>(std::move(joined));
873}

945template <typename T>
946template <typename ValueSelector>
947Queryable<T>
948Queryable<T>::order_by(ValueSelector value_selector, bool descending) {
949 static_assert(
950 std::is_copy_assignable_v<T> ||
951 (std::is_move_assignable_v<T> && std::is_move_constructible_v<T>),
952 "T must either be copy assignable, or move assignable and "
953 "constructible.");
954 static_assert(
955 traits::is_less_than_comparable<decltype(value_selector(
956 *items_.begin()))>::value,
957 "ValueSelector return type must be less-than compareable.");
958
959 using K = std::decay_t<decltype(value_selector(*items_.begin()))>;
960 // Select each value once, then stable sort the indices by value.
961 std::vector<K> values;
962 values.reserve(items_.size());
963 for (T &item : items_) {
964 values.push_back(value_selector(std::as_const(item)));
965 }
966 std::vector<size_t> order(items_.size());
967 std::iota(order.begin(), order.end(), 0);
968 std::stable_sort(
969 order.begin(), order.end(),
970 [&values, descending](size_t lhs, size_t rhs) {
971 return descending ? values[rhs] < values[lhs]
972 : values[lhs] < values[rhs];
973 });
974
975 std::vector<T> ordered;
976 ordered.reserve(items_.size());
977 for (size_t i : order) {
978 ordered.push_back(std::move(items_[i]));
979 }
980 return Queryable<T>(std::move(ordered));
981}

1024template <typename T>
1025Queryable<T>
1026Queryable<T>::slice(size_t start_index, size_t size, size_t stride) {
1027 if (size == 0) {
1028 return Queryable<T>(std::vector<T>());
1029 }
1030 asserts::invariant::eval(start_index < this->size())
1031 << "Slice start index " << start_index
1032 << " must be less than sequence size of " << this->size() << ".";
1033
1034 size_t last_index = start_index + (size - 1) * stride;
1035 asserts::invariant::eval(last_index < this->size())
1036 << "Slice last index " << last_index
1037 << " must be less than the sequence size of " << this->size() << ".";
1038
1039 std::vector<T> sliced;
1040 sliced.reserve(size);
1041 if (stride == 0) {
1042 // Every item is the same one, so all but the last must be copies.
1043 for (size_t i = 0; i + 1 < size; i++) {
1044 sliced.push_back(detail::copy_or_throw(items_[start_index], "slice"));
1045 }
1046 sliced.push_back(std::move(items_[start_index]));
1047 } else {
1048 for (size_t i = start_index; sliced.size() < size; i += stride) {
1049 sliced.push_back(std::move(items_[i]));
1050 }
1051 }
1052
1053 return Queryable<T>(std::move(sliced));
1054}

1072template <typename T>
1073Queryable<T> Queryable<T>::take_random(size_t value) {
1074 asserts::invariant::eval(value <= size())
1075 << "Take random value " << value
1076 << " must be less than or equal to sequence size of " << size() << ".";
1077
1078 // Partial Fisher-Yates shuffle: only the first value positions are drawn.
1079 auto &engine = detail::random_engine();
1080 for (size_t i = 0; i < value; i++) {
1081 std::uniform_int_distribution<size_t> dist(i, items_.size() - 1);
1082 std::swap(items_[i], items_[dist(engine)]);
1083 }
1084 items_.erase(items_.begin() + value, items_.end());
1085 return Queryable<T>(std::move(items_));

1161template <typename T>
1162template <typename U>
1163Queryable<std::tuple<T, U>>
1164Queryable<T>::zip(std::vector<U> rhs_items, bool truncate) {
1165 size_t common = std::min(items_.size(), rhs_items.size());
1166 size_t total = truncate ? common : std::max(items_.size(), rhs_items.size());
1167
1168 std::vector<std::tuple<T, U>> zipped;
1169 zipped.reserve(total);
1170 for (size_t i = 0; i < common; i++) {
1171 zipped.emplace_back(std::move(items_[i]), std::move(rhs_items[i]));
1172 }
1173 if (total == common) {
1174 return Queryable<std::tuple<T, U>>(std::move(zipped));
1175 }
1176
1177 // Pad the shorter side with default values.
1178 if (items_.size() > rhs_items.size()) {
1179 if constexpr (std::is_default_constructible_v<U>) {
1180 for (size_t i = common; i < total; i++) {
1181 zipped.emplace_back(std::move(items_[i]), U());
1182 }
1183 } else {
1184 throw std::invalid_argument(
1185 "zip padding requires a default constructible right hand type.");
1186 }
1187 } else {
1188 if constexpr (std::is_default_constructible_v<T>) {
1189 for (size_t i = common; i < total; i++) {
1190 zipped.emplace_back(T(), std::move(rhs_items[i]));
1191 }
1192 } else {
1193 throw std::invalid_argument(
1194 "zip padding requires a default constructible left hand type.");
1195 }
1196 }
1197 return Queryable<std::tuple<T, U>>(std::move(zipped));
1198}
